cmake_minimum_required(VERSION 3.16)
project(book_service LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MHD REQUIRED IMPORTED_TARGET libmicrohttpd)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
find_package(SQLite3 REQUIRED)
add_executable(book_service main.cpp)
target_link_libraries(book_service PRIVATE PkgConfig::MHD PkgConfig::JSONC SQLite::SQLite3)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_api.py $<TARGET_FILE:book_service>)
