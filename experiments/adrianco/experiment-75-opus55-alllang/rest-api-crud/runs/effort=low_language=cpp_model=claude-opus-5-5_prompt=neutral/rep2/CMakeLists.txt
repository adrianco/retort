cmake_minimum_required(VERSION 3.16)
project(books_api CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
add_library(bookapi api.cpp)
if(TARGET SQLite3::SQLite3)
  target_link_libraries(bookapi PUBLIC SQLite3::SQLite3)
else()
  target_link_libraries(bookapi PUBLIC SQLite::SQLite3)
endif()
add_executable(books_server main.cpp)
target_link_libraries(books_server bookapi)
add_executable(books_tests tests.cpp)
target_link_libraries(books_tests bookapi)
enable_testing()
add_test(NAME books_tests COMMAND books_tests)
