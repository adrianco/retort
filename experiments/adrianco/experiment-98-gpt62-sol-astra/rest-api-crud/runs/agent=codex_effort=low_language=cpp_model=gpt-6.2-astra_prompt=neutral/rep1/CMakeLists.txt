cmake_minimum_required(VERSION 3.16)
project(book_api LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Boost 1.75 CONFIG REQUIRED COMPONENTS json)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_executable(book_api main.cpp)
add_library(book_dependencies INTERFACE)
if(TARGET SQLite3::SQLite3)
  target_link_libraries(book_dependencies INTERFACE SQLite3::SQLite3)
else()
  target_link_libraries(book_dependencies INTERFACE SQLite::SQLite3)
endif()
target_link_libraries(book_dependencies INTERFACE Boost::json Threads::Threads)
target_link_libraries(book_api PRIVATE book_dependencies)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(test_transport test_transport.cpp)
  target_link_libraries(test_transport PRIVATE book_dependencies)
  add_test(NAME api COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests.py $<TARGET_FILE:test_transport>)
endif()
