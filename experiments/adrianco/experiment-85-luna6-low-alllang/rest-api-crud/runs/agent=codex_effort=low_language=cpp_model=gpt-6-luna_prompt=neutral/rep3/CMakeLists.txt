cmake_minimum_required(VERSION 3.16)
project(book_api LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED json-c)
add_executable(book_api main.cpp)
target_include_directories(book_api PRIVATE ${JSONC_INCLUDE_DIRS})
target_link_libraries(book_api PRIVATE SQLite::SQLite3 ${JSONC_LINK_LIBRARIES})
enable_testing()
add_executable(book_api_tests tests.cpp)
target_include_directories(book_api_tests PRIVATE ${JSONC_INCLUDE_DIRS})
target_link_libraries(book_api_tests PRIVATE SQLite::SQLite3 ${JSONC_LINK_LIBRARIES})
add_test(NAME book_api_tests COMMAND book_api_tests)
